# 핵심 노트 (1단계) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 카드와 연결된 두 층 구조 핵심 노트(흐름·한 줄 핵심·왜·헷갈림·암기법·비교표)를 앱에 추가하고 배포한다.

**Architecture:** 노트 내용은 `notes.json`(시대 → 주제 → 항목, 항목마다 카드 id 목록). 계산(역색인·항목 기억률·항목 카드 배치)은 `logic.js`, 화면은 `index.html`의 새 `<section id="notes">`와 `app.js`. 내용은 시대별 병렬 작성 → 조립 → 사실 확인.

**Tech Stack:** 바닐라 JS ES 모듈, `node:test`, 정적 PWA(GitHub Pages, service worker cache-first).

**Spec:** `docs/superpowers/specs/2026-10-08-notes-design.md`

## Global Constraints

- 항목: `id` /^[a-z0-9-]+$/ 전체 유일 · `key` ≤ 40자 · `why` 1~3문장 ≤ 150자 · `confuse`?·`memo`? ≤ 60자 · `table`? 첫 행 머리글, 모든 행 같은 칸 수 · `cards` ≥ 1개.
- `notes.json` 시대 순서 = `ERAS`에서 `기타` 제외. 시대마다 마지막 주제 제목 = `헷갈리는 것`. 항목의 카드는 그 시대 카드만.
- **모든 카드는 하나 이상의 항목에 연결.** 여러 항목이면 "노트 보기"는 문서 순서상 첫 항목.
- 모든 문장은 직접 작성(교재·기출 문장 인용 금지). 사실 기준 = 감수된 `cards.json`.
- `notes.json` 불러오기 실패 시 노트 버튼만 숨기고 나머지 정상.
- 배포: `version.js` → `v8`, `sw.js` FILES에 `notes.json`.

## Review Focus

- 정답을 맞힌 카드(0.5초 뒤 자동 진행)에서 "노트 보기"를 누를 수 없어야 함 → "노트 보기"는 틀렸을 때만(Task 3 화면 확인).
- 문제 화면 → 노트 → 뒤로 가기: 풀던 문제로 돌아와야 함(홈으로 가면 안 됨)(Task 3 화면 확인).
- 항목 카드 라운드를 다 풀면 같은 카드가 무한 반복되지 않고 라운드가 끝나야 함 → `buildBatch`의 `skip`(Task 1 테스트).
- 연결 카드를 아직 한 장도 안 본 항목의 기억률은 0%(NaN 아님) → `itemRecall`(Task 1 테스트).
- `notes.json`이 없거나 깨져도 앱의 나머지는 동작(Task 3 화면 확인).

```js
export function buildBatch(cards, progress, { day, era = null, type = null, ids = null, skip = new Set(), size = BATCH_SIZE }) {
  // ...
  const pool = cards.filter(c => (!era || c.era === era) && (!type || c.type === type)

```

```js
export function noteIndex(notes) {
  const idx = new Map();
  for (const { topics } of notes) for (const { items } of topics) for (const it of items)
    for (const id of it.cards) if (!idx.has(id)) idx.set(id, it);
  return idx;
}

export function itemRecall(item, progress, day) {
  return item.cards.reduce((s, id) => s + recall(progress[id], day), 0) / item.cards.length;
}
```

```js
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { ERAS } from '../logic.js';

const read = f => JSON.parse(readFileSync(new URL('../' + f, import.meta.url), 'utf8'));
const notes = read('notes.json'), cards = read('cards.json');
const byId = new Map(cards.map(c => [c.id, c]));
const items = notes.flatMap(n => n.topics.flatMap(t => t.items.map(it => ({ ...it, era: n.era }))));
const ITEM_KEYS = new Set(['id', 'head', 'key', 'why', 'confuse', 'memo', 'table', 'cards']);

test('eras in ERAS order, flow 3–6 steps, last topic is 헷갈리는 것', () => {
  assert.deepEqual(notes.map(n => n.era), ERAS.filter(e => e !== '기타'));
  for (const n of notes) {
    assert.ok(n.flow.length >= 3 && n.flow.length <= 6 && n.flow.every(s => s.trim()), n.era);
    assert.ok(n.topics.length >= 2 && n.topics.every(t => t.title && t.items.length), n.era);
    assert.equal(n.topics.at(-1).title, '헷갈리는 것', n.era);
  }
});

test('items are well-formed', () => {
  const ids = new Set();
  for (const it of items) {
    const where = `${it.era} ${it.id}`;
    assert.match(it.id, /^[a-z0-9-]+$/, where);
    assert.ok(!ids.has(it.id), `duplicate id ${where}`);
    ids.add(it.id);
    assert.ok(Object.keys(it).every(k => k === 'era' || ITEM_KEYS.has(k)), `unknown field ${where}`);
    assert.ok(it.head?.trim() && it.key?.trim() && it.key.length <= 40, `head/key ${where}`);
    assert.ok(it.why?.trim() && it.why.length <= 150, `why ${where}`);
    for (const k of ['confuse', 'memo']) assert.ok(it[k] === undefined || (it[k].trim() && it[k].length <= 60), `${k} ${where}`);
    if (it.table) assert.ok(it.table.length >= 2 && it.table.every(r => r.length === it.table[0].length && r.every(x => typeof x === 'string')), `table ${where}`);
    assert.ok(it.cards.length && new Set(it.cards).size === it.cards.length, `cards ${where}`);
    for (const id of it.cards) assert.equal(byId.get(id)?.era, it.era, `card ${id} missing or other era: ${where}`);
  }
});

test('every card is linked to a note item', () => {
  const linked = new Set(items.flatMap(it => it.cards));
  assert.deepEqual(cards.filter(c => !linked.has(c.id)).map(c => c.id), []);
});
```

```js
const el = (tag, text, cls) => { const e = document.createElement(tag); if (text != null) e.textContent = text; if (cls) e.className = cls; return e; };

// No cancelTimer(): opening a note from a wrong answer must keep the study session intact.
function renderNotes(era = null, focusId = null) {
  const day = L.dayOf();
  const body = $('notesBody');
  body.classList.toggle('hide', hideKeys);
  $('hideKeys').checked = hideKeys;
  if (!era) {
    $('notesTitle').textContent = '핵심 노트';
    body.replaceChildren(...notes.map(n => {
      const its = n.topics.flatMap(t => t.items);
      const pct = Math.round(100 * its.reduce((s, it) => s + L.itemRecall(it, progress, day), 0) / its.length);
      const b = btn('', () => renderNotes(n.era), 'eraRow');
      b.append(el('span', n.era), el('span', `${its.length}개 항목 · ${pct}%`));
      return b;
    }));
    return show('notes');
  }
  const n = notes.find(x => x.era === era);
  $('notesTitle').textContent = `${era} 노트`;
  const parts = [btn('← 시대 목록', () => renderNotes()), el('p', n.flow.join(' → '), 'flow')];
  for (const t of n.topics) {
    parts.push(el('h2', t.title));
    for (const it of t.items) parts.push(noteItem(it, day, it.id === focusId));
  }
  body.replaceChildren(...parts);
  show('notes');
  if (focusId) document.getElementById('n-' + focusId)?.scrollIntoView({ block: 'start' });
}

function noteItem(it, day, open) {
  const d = el('details');
  d.id = 'n-' + it.id;
  d.open = open;
  const key = el('span', it.key, 'key');
  key.onclick = e => { if (hideKeys && !key.classList.contains('shown')) { e.preventDefault(); key.classList.add('shown'); } };
  const s = el('summary');
  s.append(el('b', it.head), key, el('span', `${Math.round(100 * L.itemRecall(it, progress, day))}%`, 'pct'));
  d.append(s, el('p', it.why));
  if (it.confuse) d.append(el('p', '⚠ ' + it.confuse));
  if (it.memo) d.append(el('p', '🔑 ' + it.memo));
  if (it.table) {
    const tb = el('table');
    it.table.forEach((row, i) => { const tr = el('tr'); row.forEach(x => tr.append(el(i ? 'td' : 'th', x))); tb.append(tr); });
    d.append(tb);
  }
  d.append(btn(`이 항목 카드 ${it.cards.length}장 풀기`, () => { noteIds = new Set(it.cards); enterMode('note'); }, 'primary'));
  return d;
}
```
